def function(s,name):
    start=re.search(r'\bfn\s+'+name+r'\s*\(',s).start();opening=s.index('{',start);level=1;end=opening+1
    while level:
        if s[end]=='{':level+=1
        elif s[end]=='}':level-=1
        end+=1
    return s[start:end]
